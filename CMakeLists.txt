cmake_minimum_required(VERSION 3.25)
project(BiglyBTC-plugin-azlocprov VERSION 0.1.0 LANGUAGES CXX)
find_package(BiglyBTCPluginSDK 0.1.0 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MAXMIND REQUIRED IMPORTED_TARGET libmaxminddb>=1.12.2)
find_package(LibArchive 3.8 REQUIRED)
add_library(BiglyBTC-plugin-azlocprov MODULE location.cpp)
set_target_properties(BiglyBTC-plugin-azlocprov PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON CXX_EXTENSIONS OFF)
target_compile_features(BiglyBTC-plugin-azlocprov PRIVATE cxx_std_23)
target_compile_options(BiglyBTC-plugin-azlocprov PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(BiglyBTC-plugin-azlocprov PRIVATE BT_NATIVE_VERSION="${PROJECT_VERSION}")
target_link_libraries(BiglyBTC-plugin-azlocprov PRIVATE BiglyBTC::PluginAPI
  PkgConfig::MAXMIND LibArchive::LibArchive)
option(BIGLYBTC_PLUGIN_TESTS "Build generated-profile plugin tests" ON)
if(BIGLYBTC_PLUGIN_TESTS)
  find_package(ICU 78.2 REQUIRED COMPONENTS uc)
  enable_testing()
  add_executable(location-driver tests/driver.cpp)
  target_link_libraries(location-driver PRIVATE BiglyBTC::PluginHost ICU::uc)
  target_compile_features(location-driver PRIVATE cxx_std_23)
  target_compile_options(location-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
  option(BIGLYBTC_PLUGIN_UPDATE_FIXTURES "Build actual alternate-version modules for offline upgrade tests" OFF)
  if(BIGLYBTC_PLUGIN_UPDATE_FIXTURES)
    foreach(minor RANGE 2 6)
      add_library(location-update-${minor} MODULE location.cpp)
      set_target_properties(location-update-${minor} PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden
        VISIBILITY_INLINES_HIDDEN ON CXX_EXTENSIONS OFF)
      target_compile_features(location-update-${minor} PRIVATE cxx_std_23)
      target_compile_definitions(location-update-${minor} PRIVATE BT_NATIVE_VERSION="0.${minor}.0")
      target_compile_options(location-update-${minor} PRIVATE -Wall -Wextra -Wpedantic -Werror)
      target_link_libraries(location-update-${minor} PRIVATE BiglyBTC::PluginAPI PkgConfig::MAXMIND LibArchive::LibArchive)
    endforeach()
  endif()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(BIGLYBTC_MMDB_FIXTURE "" CACHE FILEPATH "Official synthetic GeoIP2-Country-Test.mmdb")
  if(BIGLYBTC_MMDB_FIXTURE)
    add_test(NAME location-compatible-files COMMAND Python3::Interpreter
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/compatibility.py" "$<TARGET_FILE:location-driver>"
      "$<TARGET_FILE:BiglyBTC-plugin-azlocprov>" "${BIGLYBTC_MMDB_FIXTURE}")
    set_tests_properties(location-compatible-files PROPERTIES TIMEOUT 60)
  endif()
endif()
